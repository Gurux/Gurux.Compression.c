cmake_minimum_required(VERSION 3.16)
project(GuruxV44 VERSION 1.0.0 LANGUAGES C)
set(GURUX_DLMS_ROOT "" CACHE PATH "Optional GuruxDLMS.c development directory for external buffer support")
set(GX_COMPRESSION_EXTERNAL_BUFFER_HEADER "" CACHE FILEPATH "Application Gurux bytebuffer.h; excludes the local buffer")
set(GX_COMPRESSION_BUFFER_LIBRARIES "" CACHE STRING "Libraries supplying the external gxByteBuffer implementation")
set(GX_COMPRESSION_CONFIG_HEADER "" CACHE FILEPATH "Optional shared allocation and buffer configuration header")
option(DLMS_IGNORE_MALLOC "Disable dynamic allocation" OFF)
option(COMPRESSION_TRACE "Include optional CGXV44TraceArgs and synchronous trace callbacks" OFF)
option(GX_V44_BUFFER_16 "Use the microcontroller 16-bit gxByteBuffer layout" OFF)
option(GX_V44_BUILD_BUFFER "Build buffer support; disable when the application supplies it" ON)
option(GX_V44_INTEROP_TESTS "Run reciprocal .NET and independent-fixture checks (requires dotnet and Python)" OFF)
set(GURUX_COMPRESSION_NET "${CMAKE_CURRENT_SOURCE_DIR}/../Gurux.Compression.Net" CACHE PATH "Reference .NET project")
# Supplying the old dependency root explicitly selects the external buffer.
if(GURUX_DLMS_ROOT AND NOT GX_COMPRESSION_EXTERNAL_BUFFER_HEADER)
  set(GX_COMPRESSION_EXTERNAL_BUFFER_HEADER "${GURUX_DLMS_ROOT}/include/bytebuffer.h")
endif()
add_library(gurux_v44 STATIC src/gxv44.c)
add_library(gurux_v44_config INTERFACE)
target_include_directories(gurux_v44_config INTERFACE "${CMAKE_CURRENT_SOURCE_DIR}/include")
if(GX_COMPRESSION_CONFIG_HEADER)
  target_compile_definitions(gurux_v44_config INTERFACE GX_COMPRESSION_CONFIG_HEADER="${GX_COMPRESSION_CONFIG_HEADER}")
endif()
if(GX_COMPRESSION_EXTERNAL_BUFFER_HEADER)
  if(NOT EXISTS "${GX_COMPRESSION_EXTERNAL_BUFFER_HEADER}")
    message(FATAL_ERROR "GX_COMPRESSION_EXTERNAL_BUFFER_HEADER must name an existing header")
  endif()
  get_filename_component(GX_EXTERNAL_INCLUDE "${GX_COMPRESSION_EXTERNAL_BUFFER_HEADER}" DIRECTORY)
  target_include_directories(gurux_v44_config SYSTEM INTERFACE "${GX_EXTERNAL_INCLUDE}")
  target_compile_definitions(gurux_v44_config INTERFACE GX_COMPRESSION_EXTERNAL_BUFFER_HEADER="${GX_COMPRESSION_EXTERNAL_BUFFER_HEADER}")
endif()
target_include_directories(gurux_v44 PUBLIC include)
set_target_properties(gurux_v44 PROPERTIES C_STANDARD 90 C_STANDARD_REQUIRED YES C_EXTENSIONS NO)
if(DLMS_IGNORE_MALLOC)
  target_compile_definitions(gurux_v44_config INTERFACE DLMS_IGNORE_MALLOC)
endif()
if(COMPRESSION_TRACE)
  target_compile_definitions(gurux_v44_config INTERFACE COMPRESSION_TRACE)
endif()
if(GX_V44_BUFFER_16)
  target_compile_definitions(gurux_v44_config INTERFACE GX_DLMS_MICROCONTROLLER)
endif()
if(MSVC)
  target_compile_options(gurux_v44 PRIVATE /W4)
else()
  target_compile_options(gurux_v44 PRIVATE -pedantic-errors -Wall -Wextra -Wpedantic -Werror -Wdeclaration-after-statement)
endif()
target_link_libraries(gurux_v44 PUBLIC gurux_v44_config)
if(GX_COMPRESSION_EXTERNAL_BUFFER_HEADER AND GX_COMPRESSION_BUFFER_LIBRARIES)
  target_link_libraries(gurux_v44 PUBLIC ${GX_COMPRESSION_BUFFER_LIBRARIES})
elseif(GX_V44_BUILD_BUFFER)
  if(GX_COMPRESSION_EXTERNAL_BUFFER_HEADER)
    # Optional convenience build of the actual external Gurux implementation.
    if(NOT GURUX_DLMS_ROOT)
      message(FATAL_ERROR "Supply GX_COMPRESSION_BUFFER_LIBRARIES, GURUX_DLMS_ROOT or set GX_V44_BUILD_BUFFER=OFF")
    endif()
    add_library(gurux_v44_buffer STATIC "${GURUX_DLMS_ROOT}/src/bytebuffer.c" "${GURUX_DLMS_ROOT}/src/helpers.c" "${GURUX_DLMS_ROOT}/src/bitarray.c")
    set_target_properties(gurux_v44_buffer PROPERTIES C_STANDARD 99 C_STANDARD_REQUIRED YES)
    target_compile_definitions(gurux_v44_buffer PRIVATE DLMS_IGNORE_STRING_CONVERTER _CRT_SECURE_NO_WARNINGS)
    if(MSVC)
      target_compile_options(gurux_v44_buffer PRIVATE /FIstdio.h)
      if(GX_COMPRESSION_CONFIG_HEADER)
        target_compile_options(gurux_v44_buffer PRIVATE "/FI${GX_COMPRESSION_CONFIG_HEADER}")
      endif()
    else()
      target_compile_options(gurux_v44_buffer PRIVATE -include stdio.h)
      if(GX_COMPRESSION_CONFIG_HEADER)
        target_compile_options(gurux_v44_buffer PRIVATE "SHELL:-include \"${GX_COMPRESSION_CONFIG_HEADER}\"")
      endif()
    endif()
    if(NOT WIN32)
      target_link_libraries(gurux_v44_buffer PUBLIC m)
    endif()
  else()
    add_library(gurux_v44_buffer STATIC src/gxbytebuffer.c)
    set_target_properties(gurux_v44_buffer PROPERTIES C_STANDARD 90 C_STANDARD_REQUIRED YES C_EXTENSIONS NO)
    if(MSVC)
      target_compile_options(gurux_v44_buffer PRIVATE /W4)
    else()
      target_compile_options(gurux_v44_buffer PRIVATE -pedantic-errors -Wall -Wextra -Werror -Wdeclaration-after-statement)
    endif()
  endif()
  target_link_libraries(gurux_v44_buffer PUBLIC gurux_v44_config)
  target_link_libraries(gurux_v44 PUBLIC gurux_v44_buffer)
endif()
include(CTest)
if(BUILD_TESTING)
  if(NOT GX_V44_BUILD_BUFFER AND NOT GX_COMPRESSION_BUFFER_LIBRARIES)
    message(FATAL_ERROR "Standalone tests need GX_V44_BUILD_BUFFER=ON. For integration set BUILD_TESTING=OFF.")
  endif()
  add_executable(v44_tests tests/test_v44.c)
  target_link_libraries(v44_tests PRIVATE gurux_v44)
  add_test(NAME v44_unit COMMAND v44_tests)
  add_executable(v44_driver tests/driver.c)
  target_link_libraries(v44_driver PRIVATE gurux_v44)
  add_executable(v44_benchmark tests/benchmark.c)
  target_link_libraries(v44_benchmark PRIVATE gurux_v44)
  add_test(NAME v44_packets COMMAND v44_benchmark)
  set(GX_TEST_TARGETS v44_tests v44_driver v44_benchmark)
  if(NOT GX_COMPRESSION_EXTERNAL_BUFFER_HEADER)
    add_executable(v44_buffer_tests tests/test_buffer.c)
    target_link_libraries(v44_buffer_tests PRIVATE gurux_v44)
    add_test(NAME v44_buffer COMMAND v44_buffer_tests)
    list(APPEND GX_TEST_TARGETS v44_buffer_tests)
    if(NOT DLMS_IGNORE_MALLOC AND NOT GX_COMPRESSION_CONFIG_HEADER)
      add_executable(v44_allocator_tests tests/test_allocator.c src/gxv44.c src/gxbytebuffer.c)
      target_link_libraries(v44_allocator_tests PRIVATE gurux_v44_config)
      target_compile_definitions(v44_allocator_tests PRIVATE GX_COMPRESSION_CONFIG_HEADER="${CMAKE_CURRENT_SOURCE_DIR}/tests/allocator_config.h")
      add_test(NAME v44_allocator COMMAND v44_allocator_tests)
      list(APPEND GX_TEST_TARGETS v44_allocator_tests)
    endif()
  endif()
  foreach(test_target IN LISTS GX_TEST_TARGETS)
    set_target_properties(${test_target} PROPERTIES C_STANDARD 90 C_STANDARD_REQUIRED YES C_EXTENSIONS NO)
    if(MSVC)
      target_compile_options(${test_target} PRIVATE /W4)
      target_compile_definitions(${test_target} PRIVATE _CRT_SECURE_NO_WARNINGS)
    else()
      target_compile_options(${test_target} PRIVATE -pedantic-errors -Wall -Wextra -Werror -Wdeclaration-after-statement)
    endif()
  endforeach()
  if(GX_V44_INTEROP_TESTS)
    find_package(Python3 REQUIRED COMPONENTS Interpreter)
    add_test(NAME v44_interop COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tests/interop.py" "$<TARGET_FILE:v44_driver>" "${GURUX_COMPRESSION_NET}")
  endif()
endif()
